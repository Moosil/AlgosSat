FUNCTION bfs(g: Graph, source: Vertex) -> Map[Vertex, Vertex]
    dist <- map with ∞ FOR EACH vertex IN g.get_vertices(); dist[source] <- 0
    visited <- ∅
    prev <- empty map
    stack <- [source]
    
    WHILE stack.size() > 0 DO
        u <- stack.pop()
        
        IF u ∉ visited THEN
            visited <- visited ∪ {u}
            
            FOR EACH v IN g.neighbours(u) DO
                w <- g.get_edge_weight(u, v)
                IF dist[u] + w < dist[v] THEN
                    prev[v] <- u
                    dist[v] <- dist[u] + w
                    stack.push(v)
                END IF
            END FOR
        END IF
    END WHILE
    RETURN prev
END FUNCTION

FUNCTION get_path_from_bfs(source: Vertex, sink: Vertex, prev: Map[Vertex, Vertex]) -> List[Vertex]
    left_path <- [source]
    right_path <- [sink]
    left <- source
    right <- sink
    
    WHILE prev.has(left) OR prev.has(right) DO
        IF prev.has(left) THEN
            left <- prev[left]
            left_path.push(left)
        END IF
        
        IF prev.has(right) THEN
            right <- prev[right]
            right_path.push(right)
        END IF
        
        FOR i <- 1 TO |left_path| DO
            IF right = left_path[i] THEN
                res <- []
                FOR j <- 1 TO i DO
                    res.push(left_path[j])
                END FOR
                
                FOR j <- 1 TO |right_path| DO
                    res.push(right_path[1 + |right_path| - j])
                END FOR
                RETURN res
            END IF
        END FOR
        
        FOR i <- 1 TO |right_path| DO
            IF left = right_path[i] THEN
                res <- []
                FOR j <- 1 TO |left_path| DO
                    res.push(left_path[j])
                END FOR
                
                FOR j <- 1 TO i DO
                    res.push(right_path[1 + i - j])
                END FOR
                RETURN res
            END IF
        END FOR
    END WHILE
    RAISE InputError
END FUNCTION

FUNCTION get_supplies_to_collect(supplies: Set[Vertex], vertex_to_supply_id: Map[Vertex, SupplyID], found_supply_ids: Set[SupplyID], supply_storage: Array[SupplyID, 5]) -> Set[Supply]
    dont_collect_supply_ids <- found_supply_ids
    FOR EACH supply IN supply_storage DO
        dont_collect_supply_ids <- dont_collect_supply_ids ∪ {supply}
    END FOR

    res <- ∅
    FOR EACH supply IN supplies DO
        IF vertex_to_supply_id[supply] ∉ dont_collect_supply_ids THEN
            res <- res ∪ {supply}
        END IF
    END FOR
    RETURN res
END FUNCTION

FUNCTION max(a: Integer, b: Integer) -> Integer
    IF a > b THEN
        RETURN a
    END IF
    RETURN b
END FUNCTION

FUNCTION get_path_length(g: Graph, path: List[Vertex]) -> Integer
    res <- 0
    FOR i <- 1 TO |path| - 1 DO
        res <- res + g.get_edge_weight(path[i], path[i + 1])
    END FOR
    RETURN res
END FUNCTION

FUNCTION reverse(list: List) -> List
    res <- []
    FOR i <- 1 TO |list| DO
        res.push(list[1 + |list| - i])
    END FOR
    RETURN res
END FUNCTION

FUNCTION reconstruct_path(prev: Map[Vertex, Vertex], sink: Vertex) -> List[Vertex]
    res <- [sink]
    WHILE prev.has(curr) DO
        curr <- prev[curr]
        res.push(curr)
    END WHILE
    RETURN reverse(res)
END FUNCTION

FUNCTION dijkstra(g: Graph, source: Vertex, sinks: Set[Vertex]) -> Map[Vertex, list[Vertex]]
    res <- empty Map
    dist <- map with ∞ FOR EACH vertex IN g.get_vertices(); dist[source] <- 0
    prev <- empty map
    
    pq <- Priority Queue
    FOR EACH vertex IN g.get_vertices() DO
        pq.enqueue(vertex, dist[vertex])
    END FOR
    WHILE |pq| > 0 DO
        u <- pq.extract_min()
        
        IF u ∈ sinks THEN
            res[u] <- reconstruct_path(prev, u)
            IF |res| = |sinks| THEN
                RETURN res
            END IF
        END IF
        
        FOR EACH v IN g.neighbours(u) DO
            w <- g.get_edge_weight(u, v)
            IF dist[u] + w < dist[v] THEN
                prev[v] <- u
                dist[v] <- dist[u] + w
                pq.update_priority(v, dist[v])
            END IF
        END FOR
    END WHILE
    RAISE InputError
END FUNCTION

FUNCTION get_path_matrix(g: Graph, entry: Vertex, exits: set[Vertex], supplies: set[Vertex]) -> Map[Vertex, Map[Vertex, List[Vertex]]]
    res <- empty Map
    FOR EACH source IN supplies ∪ {entry} DO
        res[source] <- dijkstra(g, source, supplies ∪ exits)
    END FOR
    RETURN res
END FUNCTION

FUNCTION brute_force_recursive(entry: Vertex, supplies: Set[Vertex], exits: Set[Vertex], path_cost_matrix: Map[Vertex, Map[Vertex, Integer]], fuel: Integer) -> Tuple[List[Vertex], Integer]
    min_cost <- ∞
    min_cost_path <- []
    
    IF fuel = 0 THEN
        FOR EACH exit IN exits DO
            cost <- path_cost_matrix[source][exit]
            IF cost < min_cost THEN
                min_cost <- cost
                min_cost_path <- [sink]
            END IF
        END FOR
    END IF
    IF fuel != 0 THEN
        FOR EACH sink IN sinks DO
            recursive_res <- brute_force_recursive(sink, sinks \ {sink}, exits, path_cost_matrix, fuel - 1)
            min_path_through <- recursive_res[0]
            cost <- recursive_res[1]
            cost <- cost + path_cost_matrix[source][sink]
            IF cost < min_cost THEN 
                min_cost <- cost
                min_cost_path <- [sink]
                FOR vertex IN min_path_through DO
                    min_cost_path.push(vertex)
                END FOR
            END IF
        END FOR
    END IF

END FUNCTION

FUNCTION brute_force(entry: Vertex, supplies: Set[Vertex], exits: Set[Vertex], path_cost_matrix: Map[Vertex, Map[Vertex, Integer]], fuel: Integer) -> List[Vertex]
    path <- brute_force_recursive(entry, supplies, exits, path_cost_matrix, fuel)[0]

    res <- [entry]
    FOR EACH vertex IN path DO
        res.push(path)
    END FOR
    RETURN res
END FUNCTION

FUNCTION get_F_path_from_H_path(H_path: List[Vertex], path_matrix: Map[Vertex, Map[Vertex, List[Vertex]]]) -> List[Vertex]
    res <- []
    FOR i <- 1 TO |H_path| - 1 DO
        path <- path_matrix[H_path[i]][H_path[i + 1]]
        FOR j <- 1 TO |path| - 1 DO
            res.push(path[j])
        END FOR
    END FOR
    res.push(H_path[|H_path|])
    RETURN res
END FUNCTION

FUNCTION get_which_wing(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], vertex: Vertex) -> Graph
    FOR EACH wing IN G[0] DO
        IF vertex IN g.get_vertices() THEN
            RETURN g
        END IF
    END FOR
    RAISE InputError
END FUNCTION

FUNCTION get_G_path_from_F_path(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], F_path: List[Vertex], prevs: Map[Graph, Map[VertexT, VertexT]]) -> List[Vertex]
    res <- []
    FOR i <- 1 TO |F_path| - 1 DO
        u <- F_path[i]
        v <- F_path[i + 1]
        u_wing <- get_which_wing(G, u)
        v_wing <- get_which_wing(G, v)
        IF u_wing = v_wing THEN
            path <- get_path_from_bfs(u, v, prevs[u_wing])
            FOR j <- 1 TO |path| - 1 DO
                res.push(path[j])
            END FOR
        END IF
        IF u_wing != v_wing THEN
            res.push(u)
        END IF
    END FOR
    res.push(F_path[|F_path|])
    RETURN res
END FUNCTION

FUNCTION ember_rescue(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], entry: Vertex, exits: Set[Vertex], supplies: Set[Vertex], supply_storage: Array[SupplyID, 5], vertex_to_supply_id: Map[Vertex, SupplyID], found_supply_ids: Set[SupplyID]) -> Tuple[List[Vertex], Array[Supply, 5]]
    uncollected_supplies <- get_supplies_to_collect(supplies, vertex_to_supply_id, found_supply_ids, supply_storage)
    
    num_of_supplies_to_collect <- 0
    FOR EACH id IN supply_storage DO
        IF id isn't the null id THEN
            num_of_supplies_to_collect <- num_of_supplies_to_collect + 1
        END IF
    END FOR
    num_of_supplies_to_collect <- max(num_of_supplies_to_collect, |supplies|)
    
    prevs <- empty map
    FOR EACH wing IN G[0] DO
        IF |wing.get_vertices()| > 0 THEN
            prevs[wing] <- bfs(wing, a vertex in wing)
        END IF
    END FOR
    
    F <- empty Graph
    FOR EACH v IN {entry} ∪ supplies ∪ exits DO
        F.add_vertex(v)
    END FOR
    
    junction_vertices <- ∅
    FOR EACH e IN G[1] DO
        F.add_vertex(e[0])
        F.add_vertex(e[1])
        F.add_edge(e[0], e[1], 1)
        
        junction_vertices <- junction_vertices ∪ {e[0]}
        junction_vertices <- junction_vertices ∪ {e[1]}
    END FOR
    
    FOR EACH wing IN G[0] DO
        salient_vertices <- []
        FOR EACH v IN ({entry} ∩ wing.get_vertices()) ∪ (supplies ∩ wing.get_vertices()) ∪ (exits ∩ wing.get_vertices()) ∪ (junction_vertices ∩ wing.get_vertices()) DO
            salient_vertices.push(v)
        END FOR
        
        FOR i <- 1 TO |salient_vertices| DO
            FOR j <- i + 1 TO |salient_vertices| DO
                u <- salient_vertices[i]
                v <- salient_vertices[i + 1]
                
                path <- get_path_from_bfs(u, v, prevs[wing])
                weight <- get_path_length(wing, path)
                
                F.add_edge(u, v, weight)
            END FOR
        END FOR
    END FOR
    
    path_matrix <- get_path_matrix(F, entry, exits, uncollected_supplies)
    
    path_cost_matrix <- get_path_cost_matrix(F, path_matrix)
    
    H_path <- brute_force(entry, supplies, exits, path_cost_matrix, num_of_supplies_to_collect)
    
    F_path <- get_F_path_from_H_path(H_path, path_matrix)
    
    G_path <- get_G_path_from_F_path(G, F_path, prevs)
    
    collected_supplies <- []
    FOR EACH v IN H_path DO
        IF v ∈ uncollected_supplies THEN
            collected_supplies.push(v)
        END IF
    END FOR
    
    supply_idx <- 1
    supply_storage_idx <- 1
    WHILE supply_storage_idx <= |supply_storage| AND supply_idx <= num_of_supplies_to_collect DO
        IF supply_storage[supply_storage_idx] = NULL THEN
            supply_storage[supply_storage_idx] <- vertex_to_supply_id[collected_supplies[supply_idx]]
            supply_idx <- supply_idx + 1
        END IF
    END WHILE
    RETURN G_path, supply_storage
END FUNCTION
