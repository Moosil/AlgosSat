PROCEDURE knapsack_capacity(cap: Integer, val: List[Integer], cost: List[Integer]) -> Tuple[List[Integer], Integer]:
    dp <- []
    res <- []
    FOR i <- 0 TO cap DO
        push_back(dp, 0)
        push_back(res, [])
    END FOR

    FOR i <- 1 TO length(cost) DO
        j <- cap + 1
        cost_i <- cost[i];
        WHILE j > cost_i DO
            curr <- dp[j - cost_i] + val[i]
            IF dp[j] < curr THEN
                dp[j] <- curr
                res[j] <- new_res[j]
                push_back(res[j], i)
            END IF
            j <- j - 1
        END WHILE
    END FOR

    sort(old_res[cap])
    RETURN (old_res[cap], dp[cap])
END PROCEDURE

PROCEDURE knapsack_value(cap: Integer, val: List[Integer], cost: List[Integer]) -> Tuple[List[Integer], Integer]:
    max_val <- 0
    FOR EACH v IN val DO
        max_val <- max_val + v
    END FOR

    dp <- [0]
    res <- [[]]
    FOR i <- 1 TO max_val DO
        push_back(dp, ∞)
        push_back(res, [])
    END FOR

    FOR i <- 1 TO length(val) DO
        j <- cap + 1
        val_i <- val[i];
        WHILE j > val_i DO
            curr <- dp[j - val_i] + val[i]
            IF dp[j] > curr THEN
                dp[j] <- curr
                res[j] <- res[j - val_i]
                push_back(res[j], i)
            END IF
            j <- j - 1
        END WHILE
    END FOR

    WHILE total_val >= 1 DO
        IF dp[total_val] <= cap THEN
            sort(old_res[total_val])
            RETURN (old_res[total_val], dp[total_val])
        END IF
        total_val <- total_val - 1
    END WHILE
    RETURN ([], 0)
END PROCEDURE

PROCEDURE get_path_length(g: Graph, path: List[Vertex]) -> Positive Integer or 0
    IF length(path) <= 1 THEN
	    RETURN 0
    END IF
    res <- 0
    FOR i <- 1 TO length(path) - 1 DO
        res <- res + get_edge_weight(g, path[i], path[i + 1])
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE get_which_wing(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], vertex: Vertex) -> Graph
    FOR EACH wing IN G[1] DO
        IF element_of(get_vertices(g), vertex) THEN
            RETURN g
        END IF
    END FOR
    RAISE InputError
END PROCEDURE

PROCEDURE find(l: List, x, start: Integer) -> Integer
    FOR i <- start TO length(l) DO
        IF l[i] = x THEN
            RETURN i
        END IF
    END FOR
    RETURN -1
END PROCEDURE

PROCEDURE sort(l: List) -> None
    FOR i <- 2 TO length(l) DO
        key <- l[i]
        j <- i - 1
        WHILE j >= 1 AND l[j] < key DO
            l[j + 1] <- l[j]
        END WHILE
        l[j + 1] = key
    END FOR
END PROCEDURE

PROCEDURE reverse(list: List) -> List
    res <- []
    FOR i <- 1 TO length(list) DO
        push(res, list[1 + length(list) - i])
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE reconstruct_path(prev: Map[Vertex, Vertex], sink: Vertex) -> List[Vertex]
    res <- [sink]
    curr <- sink
    WHILE has(prev, curr) DO
        curr <- prev[curr]
        push_back(res, curr)
    END WHILE
    RETURN reverse(res)
END PROCEDURE

PROCEDURE reconstruct_path_to(prev: Map[Vertex, Vertex], source: Vertex, sink: Vertex) -> List[Vertex]
    left_path <- [source]
    right_path <- [sink]
    left <- source
    right <- sink

    WHILE has(prev, left) OR has(prev, right) DO
        IF has(prev, left) THEN
            left <- prev[left]
            push_back(left_path, left)
        END IF

        IF has(prev, right) THEN
            right <- prev[right]
            push_back(right_path, right)
        END IF

        FOR i <- 1 TO length(left_path) DO
            IF right = left_path[i] THEN
                res <- []
                FOR j <- 1 TO i DO
                    push_back(res, left_path[j])
                END FOR

                FOR j <- 1 TO length(right_path) DO
                    push_back(res, right_path[1 + length(right_path) - j])
                END FOR
                RETURN res
            END IF
        END FOR

        FOR i <- 1 TO length(right_path) DO
            IF left = right_path[i] THEN
                res <- []
                FOR j <- 1 TO length(left_path) DO
                    push_back(res, left_path[j])
                END FOR

                FOR j <- 1 TO i DO
                    push_back(res, right_path[1 + i - j])
                END FOR
                RETURN res
            END IF
        END FOR
    END WHILE
    RAISE InputError
END PROCEDURE

PROCEDURE dijkstra(g: Graph, source: Vertex) -> Map[Vertex, Vertex]
    dist <- Map with ∞ FOR EACH vertex IN get_vertices(g); dist[source] <- 0
    prev <- empty Map

    pq <- empty Priority Queue
    FOR EACH vertex IN get_vertices(g) DO
        enqueue(pq, vertex, dist[vertex])
    END FOR
    WHILE length(pq) > 0 DO
        u <- extract_min(pq)

        FOR EACH v IN get_neighbours(g, u) DO
            w <- get_edge_weight(g, u, v)
            IF dist[u] + w < dist[v] THEN
                prev[v] <- u
                dist[v] <- dist[u] + w
                update_priority(pq, v, dist[v])
            END IF
        END FOR
    END WHILE
    RETURN prev
END PROCEDURE

PROCEDURE flatten_graph(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]]) -> Graph
    res <- empty Graph
    FOR EACH wing IN G[1] DO
        FOR EACH v IN get_vertices(wing) DO
            add_vertex(res, v)
        END FOR
        FOR EACH e IN get_edges(wing) DO
            add_edge(res, e[1], e[2], get_edge_weight(wing, e[1], e[2]))
        END FOR
    END FOR
    FOR EACH e IN G[2] DO
        add_edge(res, e[1], e[2], 1)
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE get_weight_cost(weight: Integer, cap: Integer, path_len: Integer) -> Integer
    RETURN ceil((1 - pow(weight / cap, 0.45)) * (cap + 2) * weight / cap + pow(weight / cap, 0.45) * (weight + 2) * path_len)
END PROCEDURE

PROCEDURE get_reduced_supplies(g: Graph, supplies: Set[Vertex], supply_weight: Map[Vertex, Integer], supply_value: Map[Vertex, Integer], entry_path: Map[Vertex, List[Vertex]], budget: Integer, drone_capacity: Integer) -> List[Vertex]
    supply_value_ordered <- []
    supply_cost_ordered <- []
    supplies_ordered <- []
    FOR EACH v IN supplies DO
        IF supply_weight[v] <= drone_capacity THEN
            push_back(supply_value_ordered, supply_value[v])
            push_back(supply_cost_ordered, get_weight_cost(supply_weight[v], drone_capacity, get_path_length(g, entry_path[v])))
            push_back(supplies_ordered, v)
        END IF
    END FOR

    res <- []
    FOR EACH i IN knapsack_value(ceil(budget), supply_value_ordered, supply_cost_ordered)[1] DO
        push_back(res, supplies_ordered[i])
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE get_other_junction(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], v: Vertex) -> Vertex
    FOR EACH je in G[2] DO
        IF je[1] = v THEN
            RETURN je[2]
        END IF
        IF je[2] = v THEN
            RETURN je[1]
        END IF
    END FOR
    RETURN v
END PROCEDURE

PROCEDURE get_supplies_to_collect(supplies: Set[Vertex], vertex_to_supply_id: Map[Vertex, SupplyID], found_supply_ids: Set[SupplyID]) -> Set[Supply]
    res <- ∅
    FOR EACH supply IN supplies DO
        IF NOT element_of(found_supply_ids, vertex_to_supply_id[supply]) THEN
            insert(res, supply)
        END IF
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE get_supply_wing_paths(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], supplies: List[Vertex], entry_to_supply: Map[Vertex, List[Vertex]]) -> Map[List[Vertex], Set[Int]]
    junctions <- {}
    FOR EACH j IN G[2] DO
        insert(junctions, j[1])
        insert(junctions, j[2])
    END FOR

    res <- empty Map
    FOR s_idx <- 1 TO length(supplies) DO
        junction_path <- []
        curr_path <- entry_to_supply[supplies[s_idx]]
        curr <- curr_path[1]
        next <- curr_path[2]
        FOR i <- 1 TO length(curr_path) - 1 DO
            IF element_of(junctions, curr) AND next = get_other_junction(G, curr) THEN
                push_back(junction_path, curr)
                push_back(junction_path, next)
            END IF
            curr <- next
            next <- curr_path[i + 2]
        END FOR

        IF has(res, junction_path) THEN
            insert(res[junction_path], s_idx)
        ELSE
            res[junction_path] <- {s_idx}
        END IF
    END FOR

    RETURN res
END PROCEDURE

PROCEDURE knapsack_supplies(supplies: List[Vertex], supply_weight: List[Integer], supply_value: List[Integer], supplies_in_junction: List[Integer], drone_capacity: Integer, entry: Vertex, backtrack: List[Vertex], res: List[Tuple[Vertex, Integer, Integer, Integer]]) -> NULL
    total_weight <- 0
    FOR EACH s_idx IN supplies_in_junction DO
        total_weight <- total_weight + supply_weight[s_idx]
    END FOR

    WHILE total_weight >= drone_capacity DO
        in_junction_weight <- []
        FOR EACH s_idx IN supplies_in_junction DO
            push_back(in_junction_weight, supply_weight[s_idx])
        END FOR
        sack <- knapsack_capacity(drone_capacity, in_junction_weight, in_junction_weight)

        sack_items <- {}
        FOR EACH ij_idx IN sack[1] DO
            in_junction_idx <- supplies_in_junction[ij_idx]
            curr_supply <- supplies[in_junction_idx]
            IF has(sack_items, curr_supply) THEN
                push_back(sack_items[curr_supply], in_junction_idx)
            ELSE
                sack_items[curr_supply] <- [in_junction_idx]
            END IF
        END FOR
        total_sack_weight <- 0
        extra_supplies <- []
        collecting_strays <- False
        i <- length(backtrack)
        WHILE i >= 1 DO
            curr_pos <- backtrack[i]
            IF has(sack_items, curr_pos) THEN
                collecting_strays <- True
                FOR s_idx IN sack_items[curr_pos] DO
                    w <- supply_weight[s_idx]
                    drop_count <- 1
                    WHILE total_sack_weight + w > drone_capacity DO
                        del_s_idx <- extra_supplies[length(extra_supplies)]
                        pop_back(extra_supplies)
                        del_pos <- backtrack[i + drop_count]
                        WHILE del_pos != supplies[del_s_idx] && find(supplies, del_pos, 0) != -1 DO
                            drop_count <- drop_count + 1
                            del_pos <- backtrack[i + drop_count]
                        END WHILE
                        del_w <- supply_weight[del_s_idx]
                        push_back(res, (del_pos, 2, del_w, supply_value[del_s_idx]))
                        total_sack_weight <- total_sack_weight - del_w
                        supplies[del_s_idx] <- del_pos
                        drop_count <- drop_count + 1
                    END WHILE

                    total_sack_weight <- total_sack_weight + w
                    push_back(res, (curr_pos, 3, w, supply_value[s_idx]))
                END FOR
            ELSE
                IF collecting_strays THEN
                    s_idx <- find(supplies, curr_pos, 0)
                    WHILE s_idx != -1 DO
                        IF find(supplies_in_junction, s_idx, 0) != -1 THEN
                            w <- supply_weight[s_idx]
                            IF total_sack_weight + w <= drone_capacity THEN
                                total_sack_weight <- total_sack_weight + w
                                push_back(res, (curr_pos, 3, w, supply_value[s_idx]))
                                push_back(extra_supplies, s_idx)
                            END IF
                        END IF
                        s_idx <- find(supplies, curr_pos, s_idx)
                    END WHILE
                END IF
            END IF
            i <- i - 1
        END WHILE

        FOR i IN sack[1] DO
            s_idx <- supplies_in_junction[i]
            supplies[s_idx] <- entry
            pop_at(supplies_in_junction, i)
            w <- supply_weight[s_idx]
            total_weight <- total_weight - w
            push_back(res, (entry, 2, w, supply_value[s_idx]))
        END FOR
        FOR s_idx IN extra_supplies DO
            supplies[s_idx] <- entry
            pop_at(supplies_in_junction, find(supplies_in_junction, s_idx, 0))
            w <- supply_weight[s_idx]
            total_weight <- total_weight - w
            push_back(res, (entry, 2, w, supply_value[s_idx]))
        END FOR
    END WHILE
END PROCEDURE

PROCEDURE clear_junction_path(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], entry: Vertex, curr: Vertex, backtrack: list[Vertex], supplies: List[Vertex], supply_weight: List[Integer], supply_value: List[Integer], supply_path: Map[List[Vertex], Set[Vertex]], inter_wing_path: List[Vertex], drone_capacity: Integer, res: List[Tuple[Vertex, Integer, Integer, Integer]]) -> NULL
    junction_other <- get_other_junction(G, curr)
    IF junction_other = curr THEN
        RETURN
    END IF

    new_inter_wing_path <- inter_wing_path
    push_back(new_inter_wing_path, curr)
    push_back(new_inter_wing_path, junction_other)

    worth_doing <- False
    min_length <- length(new_inter_wing_path)
    FOR EACH k IN get_keys(supply_path) DO
        worth_doing <- True
        IF length(k) >= min_length THEN
            FOR i <- 1 TO min_length DO
                IF k[i] != new_inter_wing_path[i] THEN
                    worth_doing <- False
                    BREAK
                END IF
            END FOR
        END IF
        IF worth_doing THEN
            BREAK
        END IF
    END FOR

    IF worth_doing THEN
        push_back(res, (curr, 1, 0, 0))
        curr_backtrack <- backtrack
        push_back(curr_backtrack, junction_other)

        wing <- get_which_wing(G, junction_other)
        surviving_supplies <- []
        FOR EACH n IN get_neighbours(wing, junction_other) DO
            new_branch_backtrack <- [junction_other]
            branch_res <- clear_branch(G, entry, junction_other, n, wing, new_branch_backtrack, supplies, supply_weight, supply_value, supply_path, new_inter_wing_path, drone_capacity)

            IF length(branch_res) != 0 THEN
                supplies_in_junction <- []
                FOR EACH s_idx in supply_path[new_inter_wing_path] DO
                    IF supplies[s_idx] == junction_other THEN
                        push_back(supplies_in_junction, s_idx)
                    END IF
                END FOR

                bt_i <- length(backtrack)
                i <- length(branch_res)
                WHILE i >= 1 DO
                    instruction <- branch_res[i]
                    IF instruction[1] = junction_other && instruction[2] = 2 THEN
                        w <- instruction[3]
                        v <- instruction[4]
                        FOR j <- 1 TO length(supplies_in_junction) DO
                            s_idx <- supplies_in_junction[j]
                            IF supplies[s_idx] = junction_other && supply_weight[s_idx] = w && supply_value[s_idx] = v THEN
                                curr_bt <- backtrack[bt_i]
                                supplies[bt_i] <- curr_bt
                                supplies_in_junction.pop_at(i)
                                surviving_supplies.push_back(s_idx)
                                branch_res[i] <- (curr_bt, 2, w, v)
                                IF bt_i > 1 THEN
                                    bt_i <- bt_i - 1
                                END IF
                                BREAK
                            END IF
                        END FOR
                    END IF
                END WHILE
            END IF
        END FOR

        knapsack_supplies(supplies, supply_weight, supply_value, surviving_supplies, drone_capacity, entry, backtrack, res)

        IF has(supply_path, new_inter_wing_path) THEN
            supply_path[inter_wing_path] <- union(supply_path[inter_wing_path], supply_path[new_inter_wing_path])
            remove(supply_path, new_inter_wing_path)
        END IF
    END FOR
END PROCEDURE

PROCEDURE clear_branch(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], entry: Vertex, orig: Vertex, branch: Vertex, orig_wing: Graph, backtrack: list[Vertex], supplies: List[Vertex], supply_weight: List[Integer], supply_value: List[Integer], supply_path: Map[List[Vertex], Set[Vertex]], inter_wing_path: List[Vertex], drone_capacity: Integer) -> List[Tuple[Vertex, Integer, Integer, Integer]]
    res <- []

    branch_pos <- branch
    prev <- orig
    push_back(backtrack, branch_pos)

    clear_junction_path(G, entry, branch_pos, backtrack, supplies, supply_weight, supply_value, supply_path, inter_wing_path, drone_capacity, res)
    neighbours <- get_neighbours(orig_wing, branch_pos)
    WHILE length(neighbours) = 2 DO
        IF neighbours[1] = prev THEN
            prev <- branch_pos
            branch_pos <- neighbours[2]
        ELSE
            prev <- branch_pos
            branch_pos <- neighbours[1]
        END IF
        neighbours <- get_neighbours(orig_wing, branch_pos)
        push_back(backtrack, branch_pos)

        clear_junction_path(G, entry, branch_pos, backtrack, supplies, supply_weight, supply_value, supply_path, inter_wing_path, drone_capacity, res)
    END WHILE

    branch_degree <- length(neighbours)
    IF branch_degree >= 3 THEN
        FOR EACH n IN neighbours DO
            IF n != prev THEN
                new_backtrack <- []
                FOR EACH i IN backtrack DO
                    push_back(new_backtrack, i)
                END FOR
                FOR EACH i IN clear_branch(G, entry, branch_pos, n, orig_wing, new_backtrack, supplies, supply_weight, supply_value, supply_path, inter_wing_path, drone_capacity) DO
                    push_back(res, i)
                END FOR
            END IF
        END FOR
    END IF

    backtrack_set <- {}
    FOR EACH i IN backtrack DO
        insert(backtrack_set, i)
    END FOR

    supply_to_collect <- []
    IF has(supply_path, inter_wing_path) THEN
        curr_supply_path <- supply_path[inter_wing_path]
        FOR s_idx <- 1 TO length(supplies) DO
            IF supplies[s_idx] != entry AND element_of(backtrack_set, supplies[s_idx]) AND element_of(curr_supply_path, s_idx) THEN
                push_back(supply_to_collect, s_idx)
            END IF
        END FOR
    END IF

    knapsack_supplies(supplies, supply_weight, supply_value, supply_to_collect, drone_capacity, entry, backtrack, res)

    storage <- []
    vertex_to_collect <- empty Map
    FOR EACH s_idx IN supply_to_collect DO
        v <- supplies[s_idx]
        IF has(vertex_to_collect, v) THEN
            push_back(vertex_to_collect[v], s_idx)
        ELSE
            vertex_to_collect[v] <- [s_idx]
        END IF
    END FOR

    curr_pos <- branch_pos
    res_len <- length(res)
    IF res_len != 0 THEN
        res_last <- res[res_len]
        IF res_last[1] != entry AND res_last[2] = 2 THEN
            curr_pos <- res_last[1]

            IF curr_pos != branch_pos AND curr_pos != entry THEN
                WHILE res_last[1] = curr_pos DO
                    pop_back(res)
                    res_len <- res_len - 1

                    candidates <- vertex_to_collect[curr_pos]
                    k <- length(candidates)
                    WHILE k >= 1 DO
                        s_idx <- candidates[k]
                        IF supply_weight[s_idx] = res_last[3] AND supply_value[s_idx] = res_last[4] THEN
                            push_back(storage, s_idx)
                            pop_at(candidates, k)
                            BREAK
                        END IF
                        k <- k - 1
                    END WHILE
                    res_last <- res[res_len]
                END WHILE

                WHILE res_len >= 1 DO
                    IF res_last[2] = 3 THEN
                        BREAK
                    END IF
                    IF res_last[2] = 2 THEN
                        candidates <- vertex_to_collect[res_last[0]]
                        pop_back(res)
                        res_len <- res_len - 1
                        k <- length(candidates)
                        WHILE k >= 1 DO
                            s_idx <- candidates[k]
                            IF supply_weight[s_idx] = res_last[3] AND supply_value[s_idx] = res_last[4] THEN
                                push_back(storage, s_idx)
                                pop_at(candidates, k)
                            END IF
                            k <- k - 1
                        END WHILE
                    END IF
                    res_last <- res[res_len]
                END WHILE
            END IF
        END IF
    END IF

    i <- length(backtrack)
    WHILE i >= 1 DO
        curr_pos <- backtrack[i]
        IF curr_pos = orig THEN
            BREAK
        END IF

        IF has(vertex_to_collect, curr_pos) THEN
            curr_pos_supplies <- vertex_to_collect[curr_pos]
            j <- length(curr_pos_supplies)
            WHILE j >= 1 DO
                IF find(storage, curr_pos_supplies[j], 0) = -1 THEN
                    s_idx <- curr_pos_supplies[j]
                    pop_at(curr_pos_supplies, j)
                    push_back(storage, s_idx)
                    push_back(res, (curr_pos, 3, supply_weight[s_idx], supply_value[s_idx]))
                END IF
                j <- j - 1
            END WHILE
        END IF

        i <- i - 1
    END WHILE

    supplies_set <- {}
    FOR EACH s IN supplies DO
        insert(supplies_set, s)
    END FOR

    storage_len <- length(storage)
    WHILE i >= 1 AND storage_len != 0 DO
        curr_pos <- backtrack[i]
        IF NOT element_of(supplies_set, curr_pos) THEN
            s_idx <- storage[storage_len]
            pop_back(storage)
            storage_len <- storage_len - 1
            supplies[s_idx] <- curr_pos
            push_back(res, (backtrack[i], 2, supply_weight[s_idx], supply_value[s_idx]))
        END IF
        i <- i - 1
    END WHILE

    IF storage_len != 0 THEN
        curr_backtrack <- backtrack[1]
        WHILE storage_len >= 1 DO
            s_idx <- storage[storage_len]
            supplies[s_idx] <- curr_backtrack
            push_back(res, (curr_backtrack, 2, supply_weight[s_idx], supply_value[s_idx]))
            storage_len <- storage_len - 1
        END WHILE
    END IF

    RETURN res
END PROCEDURE

FUNCTION ember_rescue(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], entry: Vertex, exits: Set[Vertex], supplies: Set[Vertex], supply_weight: Map[Vertex, Integer], supply_value: Map[Vertex, Integer], budget: Integer, drone_capacity: Integer, vertex_to_supply_id: Map[Vertex, SupplyID], found_supply_ids: set[SupplyID]) -> List[Tuple[Vertex, Integer, Integer, Integer]]
    supplies <- get_supplies_to_collect(supplies, vertex_to_supply_id, found_supply_ids)

    flat_G <- flatten_graph(G)
    entry_prev <- dijkstra(flat_G, entry)
    entry_path <- {}
    FOR EACH v IN supplies DO
        entry_path[v] <- reconstruct_path(entry_prev, v)
    END FOR

    exit_run <- []
    exit_run_cost <- -1
    FOR EACH e IN exits DO
        curr <- reconstruct_path(entry_prev, e)
        curr_cost <- get_path_length(flat_G, curr)
        IF curr_cost < exit_run_cost OR exit_run_cost = -1 THEN
            exit_run <- curr
            exit_run_cost <- curr_cost
        END IF
    END FOR

    reduced_supplies <- get_reduced_supplies(flat_G, supplies, supply_weight, supply_value, entry_path, budget - exit_run_cost, drone_capacity)
    FOR EACH s IN reduced_supplies DO
        insert(found_supply_ids, vertex_to_supply_id[s])
    END FOR

    supply_wing_paths <- get_supply_wing_paths(G, reduced_supplies, entry_path)

    entry_wing <- get_which_wing(G, entry)
    supply_weight_ordered <- []
    supply_value_ordered <- []
    FOR EACH s IN reduced_supplies DO
        push_back(supply_weight_ordered, supply_weight[s])
        push_back(supply_value_ordered, supply_value[s])
    END FOR

    res <- []
    prev_pos <- entry
    FOR EACH n IN get_neighbours(flat_G, entry) DO
        backtrack <- [entry]
        curr_supply_wing_paths <- supply_wing_paths
        FOR EACH i IN clear_branch(G, entry, entry, n, entry_wing, backtrack, reduced_supplies, supply_weight_ordered, supply_value_ordered, curr_supply_wing_paths, [], drone_capacity) DO
            curr_pos <- i[1]
            IF curr_pos != prev_pos THEN
                FOR EACH v IN reconstruct_path_to(entry_prev, prev_pos, curr_pos) DO
                    push_back(res, (v, 1, 0, 0))
                END FOR
                prev_pos <- curr_pos
            END FOR
            push_back(res, i)
        END FOR
    END FOR

    IF prev_pos != entry THEN
        FOR EACH v IN reconstruct_path_to(entry_prev, prev_pos, entry) DO
            push_back(res, (v, 1, 0, 0))
        END FOR
    END IF
    FOR EACH v IN exit_run DO
        push_back(res, (v, 1, 0, 0))
    END FOR

    RETURN res
END FUNCTION
