PROCEDURE dfs(g: Graph, source: Vertex) -> Map[Vertex, Vertex]
    visited <- ∅
    prev <- empty Map
    stack <- [source]

    WHILE length(stack) > 0 DO
        u <- pop(stack)

        FOR EACH v IN get_neighbours(g, u) DO
            IF NOT element_of(visited, u) THEN
                visited <- union(visited, {u})
                prev[v] <- u
                push(stack, v)
            END IF
        END FOR
    END WHILE
    RETURN prev
END PROCEDURE

PROCEDURE get_path_from_dfs(source: Vertex, sink: Vertex, prev: Map[Vertex, Vertex]) -> List[Vertex]
    left_path <- [source]
    right_path <- [sink]
    left <- source
    right <- sink

    WHILE has(prev, left) OR has(prev, right) DO
        IF has(prev, left) THEN
            left <- prev[left]
            push(left_path, left)
        END IF

        IF has(prev, right) THEN
            right <- prev[right]
            push(right_path, right)
        END IF

        FOR i <- 1 TO length(left_path) DO
            IF right = left_path[i] THEN
                res <- []
                FOR j <- 1 TO i DO
                    push(res, left_path[j])
                END FOR

                FOR j <- 1 TO length(right_path) DO
                    push(res, right_path[1 + length(right_path) - j])
                END FOR
                RETURN res
            END IF
        END FOR

        FOR i <- 1 TO length(right_path) DO
            IF left = right_path[i] THEN
                res <- []
                FOR j <- 1 TO length(left_path) DO
                    push(res, left_path[j])
                END FOR

                FOR j <- 1 TO i DO
                    push(res, right_path[1 + i - j])
                END FOR
                RETURN res
            END IF
        END FOR
    END WHILE
    RAISE InputError
END PROCEDURE

PROCEDURE get_supplies_to_collect(supplies: Set[Vertex], vertex_to_supply_id: Map[Vertex, SupplyID], found_supply_ids: Set[SupplyID], supply_storage: Array[SupplyID, 5]) -> Set[Supply]
    dont_collect_supply_ids <- found_supply_ids
    FOR EACH supply IN supply_storage DO
        dont_collect_supply_ids <- union(dont_collect_supply_ids, {supply})
    END FOR

    res <- ∅
    FOR EACH supply IN supplies DO
        IF NOT element_of(dont_collect_supply_ids, vertex_to_supply_id[supply]) THEN
            res <- union(res, {supply})
        END IF
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE max(a: Real, b: Real) -> Real
    IF a > b THEN
        RETURN a
    END IF
    RETURN b
END PROCEDURE

PROCEDURE get_path_length(g: Graph, path: List[Vertex]) -> Positive Integer or 0
    IF length(path) <= 1 THEN
	    RETURN 0
    END IF
    res <- 0
    FOR i <- 1 TO length(path) - 1 DO
        res <- res + get_edge_weight(g, path[i], path[i + 1])
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE reverse(list: List) -> List
    res <- []
    FOR i <- 1 TO length(list) DO
        push(res, list[1 + length(list) - i])
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE reconstruct_path(prev: Map[Vertex, Vertex], sink: Vertex) -> List[Vertex]
    res <- [sink]
    WHILE has(prev, curr) DO
        curr <- prev[curr]
        push(res, curr)
    END WHILE
    RETURN reverse(res)
END PROCEDURE

PROCEDURE dijkstra(g: Graph, source: Vertex, sinks: Set[Vertex]) -> Map[Vertex, List[Vertex]]
    res <- empty Map
    dist <- Map with ∞ FOR EACH vertex IN get_vertices(g); dist[source] <- 0
    prev <- empty Map

    pq <- empty Priority Queue
    FOR EACH vertex IN get_vertices(g) DO
        enqueue(pq, vertex, dist[vertex])
    END FOR
    WHILE length(pq) > 0 DO
        u <- extract_min(pq)

        IF element_of(sinks, u) THEN
            res[u] <- reconstruct_path(prev, u)
            IF length(res) = length(sinks) THEN
                RETURN res
            END IF
        END IF

        FOR EACH v IN get_neighbours(g, u) DO
            w <- get_edge_weight(g, u, v)
            IF dist[u] + w < dist[v] THEN
                prev[v] <- u
                dist[v] <- dist[u] + w
                update_priority(pq, v, dist[v])
            END IF
        END FOR
    END WHILE
    RAISE InputError
END PROCEDURE

PROCEDURE get_path_matrix(g: Graph, entry: Vertex, exits: set[Vertex], supplies: set[Vertex]) -> Map[Vertex, Map[Vertex, List[Vertex]]]
    res <- empty Map
    FOR EACH source IN union(supplies, {entry}) DO
        res[source] <- dijkstra(g, source, union(supplies, exits))
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE get_path_cost_matrix(g: Graph, path_matrix: Map[Vertex, Map[Vertex, List[Vertex]]]) -> Map[Vertex, Map[Vertex, Positive Integer or 0]]
    res <- empty Map
    FOR EACH source IN get_keys(path_matrix) DO
        res[source] <- empty Map
        sink_paths <- path_matrix[source]
        FOR EACH sink IN get_keys(sink_paths) DO
            res[source][sink] <- get_path_length(g, sink_paths[sink])
        END FOR
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE dp_recursive(entry: Vertex, supplies: Set[Vertex], exits: Set[Vertex], path_cost_matrix: Map[Vertex, Map[Vertex, Positive Integer or 0]], fuel: Positive Integer or 0, memo: Map[Tuple[Vertex, Set[Vertex]], Tuple[List[Vertex], Positive Integer or 0]) -> Tuple[List[Vertex], Positive Integer or 0]
    key <- (entry, supplies)
    IF has(memo, key) THEN
        RETURN memo[key]
    END IF

    min_cost <- ∞
    min_cost_path <- []

    IF fuel = 0 THEN
        FOR EACH exit IN exits DO
            cost <- path_cost_matrix[source][exit]
            IF cost < min_cost THEN
                min_cost <- cost
                min_cost_path <- [sink]
            END IF
        END FOR
    END IF
    IF fuel != 0 THEN
        FOR EACH sink IN sinks DO
            recursive_res <- dp_recursive(sink, difference(sinks, {sink}), exits, path_cost_matrix, fuel - 1)
            min_path_through <- recursive_res[0]
            cost <- recursive_res[1]
            cost <- cost + path_cost_matrix[source][sink]
            IF cost < min_cost THEN
                min_cost <- cost
                min_cost_path <- [sink]
                FOR EACH vertex IN min_path_through DO
                    push(min_cost_path, vertex)
                END FOR
            END IF
        END FOR
    END IF
    memo[key] = (min_cost_walk, min_cost)
    RETURN (min_cost_walk, min_cost)
END PROCEDURE

PROCEDURE dp(entry: Vertex, supplies: Set[Vertex], exits: Set[Vertex], path_cost_matrix: Map[Vertex, Map[Vertex, Positive Integer or 0]], fuel: Positive Integer or 0) -> List[Vertex]
    path <- dp_recursive(entry, supplies, exits, path_cost_matrix, fuel, empty Map)[0]

    res <- [entry]
    FOR EACH vertex IN path DO
        push(res, path)
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE get_F_path_from_H_path(H_path: List[Vertex], path_matrix: Map[Vertex, Map[Vertex, List[Vertex]]]) -> List[Vertex]
    res <- []
    FOR i <- 1 TO length(H_path) - 1 DO
        path <- path_matrix[H_path[i]][H_path[i + 1]]
        FOR j <- 1 TO length(path) - 1 DO
            push(res, path[j])
        END FOR
    END FOR
    push(res, H_path[length(H_path)])
    RETURN res
END PROCEDURE

PROCEDURE get_which_wing(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], vertex: Vertex) -> Graph
    FOR EACH wing IN G[0] DO
        IF element_of(get_vertices(g), vertex) THEN
            RETURN g
        END IF
    END FOR
    RAISE InputError
END PROCEDURE

PROCEDURE get_G_path_from_F_path(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], F_path: List[Vertex], prevs: Map[Graph, Map[Vertex, Vertex]]) -> List[Vertex]
    res <- []
    FOR i <- 1 TO length(F_path) - 1 DO
        u <- F_path[i]
        v <- F_path[i + 1]
        u_wing <- get_which_wing(G, u)
        v_wing <- get_which_wing(G, v)
        IF u_wing = v_wing THEN
            path <- get_path_from_dfs(u, v, prevs[u_wing])
            FOR j <- 1 TO length(path) - 1 DO
                push(res, path[j])
            END FOR
        END IF
        IF u_wing != v_wing THEN
            push(res, u)
        END IF
    END FOR
    push(res, F_path[length(F_path)])
    RETURN res
END PROCEDURE

PROCEDURE get_F(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], entry: Vertex, supplies: Set[Vertex], exits: Set[Vertex], prevs: Map[Vertex, Vertex]) -> Graph
    res <- empty Graph
    FOR EACH v IN union(union(supplies, exits), {entry}) DO
        add_vertex(res, v)
    END FOR

    junction_vertices <- ∅
    FOR EACH e IN G[1] DO
        add_vertex(res, e[0])
        add_vertex(res, e[1])
        add_edge(res, e[0], e[1], 1)

        junction_vertices <- union(junction_vertices, {e[0]})
        junction_vertices <- union(junction_vertices, {e[1]})
    END FOR

    FOR EACH wing IN G[0] DO
        salient_vertices <- []
        FOR EACH v IN intersection(get_vertices(wing), union(exits, union(supplies, union(junction_vertices, {entry})))) DO
            push(salient_vertices, v)
        END FOR

        FOR i <- 1 TO length(salient_vertices) DO
            FOR j <- i + 1 TO length(salient_vertices) DO
                u <- salient_vertices[i]
                v <- salient_vertices[i + 1]

                path <- get_path_from_dfs(u, v, prevs[wing])
                weight <- get_path_length(wing, path)

                add_edge(res, u, v, weight)
            END FOR
        END FOR
    END FOR
    RETURN res
END PROCEDURE

PROCEDURE get_new_supply_storage(supplies: Set[Vertex], supply_storage: Array[SupplyID, 5], vertex_to_supply_id: Map[Vertex, SupplyID], num_of_supplies_to_collect: Positive Integer, H_path: List[Vertex]) -> Array[SupplyID, 5]
    collected_supplies <- []
    FOR EACH v IN H_path DO
        IF element_of(supplies, v) THEN
            push(collected_supplies, v)
        END IF
    END FOR

    supply_storage_idx <- 1
    FOR i <- 1 TO num_of_supplies_to_collect DO
        IF supply_storage_idx <= length(supply_storage) THEN
            RETURN supply_storage
        END IF
        IF supply_storage[supply_storage_idx] = NULL THEN
            supply_storage[supply_storage_idx] <- vertex_to_supply_id[collected_supplies[i]]
        END IF
    END FOR
    RETURN supply_storage
END PROCEDURE

FUNCTION ember_rescue(G: Tuple[Set[Graph], Set[Tuple[Vertex, Vertex]]], entry: Vertex, supplies: Set[Vertex], exits: Set[Vertex], supply_storage: Array[SupplyID, 5], vertex_to_supply_id: Map[Vertex, SupplyID], found_supply_ids: Set[SupplyID]) -> Tuple[List[Vertex], Array[Supply, 5]]
    uncollected_supplies <- get_supplies_to_collect(supplies, vertex_to_supply_id, found_supply_ids, supply_storage)

    num_of_supplies_to_collect <- 0
    FOR EACH id IN supply_storage DO
        IF NOT id = NULL THEN
            num_of_supplies_to_collect <- num_of_supplies_to_collect + 1
        END IF
    END FOR
    num_of_supplies_to_collect <- max(num_of_supplies_to_collect, size(uncollected_supplies))

    prevs <- empty Map
    FOR EACH wing IN G[0] DO
        IF size(get_vertices(wing)) > 0 THEN
            prevs[wing] <- dfs(wing, get_vertices(wing)[0])
        END IF
    END FOR

    F <- get_F(G, entry, uncollected_supplies, exits, prevs)

    path_matrix <- get_path_matrix(F, entry, exits, uncollected_supplies)

    path_cost_matrix <- get_path_cost_matrix(F, path_matrix)

    H_path <- dp(entry, uncollected_supplies, exits, path_cost_matrix, num_of_supplies_to_collect)

    F_path <- get_F_path_from_H_path(H_path, path_matrix)

    G_path <- get_G_path_from_F_path(G, F_path, prevs)

    new_supply_storage <- get_new_supply_storage(supply_storage, vertex_to_supply_id, num_of_supplies_to_collect, H_path)
    RETURN G_path, new_supply_storage
END FUNCTION
